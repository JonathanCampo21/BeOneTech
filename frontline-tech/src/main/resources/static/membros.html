async function carregarSidebar() { try { // 1. Puxa os dados básicos que temos certeza que vêm do Login const userLogado = JSON.parse(localStorage.getItem('usuario')); if (!userLogado) return; // 2. Renderiza o nome e o cargo INSTANTANEAMENTE (nunca mais fica travado!) let fotoFinal = userLogado.fotoPerfil || `https://ui-avatars.com/api/?name=${encodeURIComponent(userLogado.nome || 'User')}&background=222&color=10b981`; const avatarElem = document.getElementById('sidebarAvatar'); const nameElem = document.getElementById('sidebarUserName'); const roleElem = document.getElementById('sidebarUserRole'); if (avatarElem) avatarElem.src = fotoFinal; if (nameElem) { // Se for a tela de usuários (que é só sua), mantém o nome puro sem Pr. if (window.location.pathname.includes('usuarios.html')) { nameElem.innerText = userLogado.nome ? userLogado.nome.split(' ')[0] : 'DEV'; } else { let primeiroNome = userLogado.nome ? userLogado.nome.split(' ')[0] : 'Visitante'; if (String(userLogado.cargo).toUpperCase().trim() === 'PASTOR') { primeiroNome = "Pr. " + primeiroNome; } nameElem.innerText = primeiroNome; } } if (roleElem) roleElem.innerText = userLogado.cargo || 'Membro'; // 3. ATUALIZAÇÃO INTELIGENTE EM SEGUNDO PLANO const res = await fetch('/api/membros'); if (res.ok) { const membros = await res.json(); // Busca das duas formas: tenta por ID, se não tiver, pareia pelo Nome! let userAtualizado = null; if (userLogado.id) { userAtualizado = membros.find(m => m.id === userLogado.id); } else if (userLogado.nome) { userAtualizado = membros.find(m => m.nome.trim().toLowerCase() === userLogado.nome.trim().toLowerCase()); } if (userAtualizado) { // Se achou a foto no banco, atualiza ela de mansinho na tela if (userAtualizado.fotoPerfil && avatarElem) { avatarElem.src = userAtualizado.fotoPerfil; } // Salva o objeto completo (agora com o ID garantido) no localStorage localStorage.setItem('usuario', JSON.stringify(userAtualizado)); } } } catch(e) { console.log("Erro ao sincronizar sidebar com o banco de dados. Usando cache local."); } }
